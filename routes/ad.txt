router.post("/save_category", async (req, res) => {
    let file_name = "";

    if (req.files && req.files.category_image) {

    res.redirect("/admin/category");
});

router.get("/edit_product/:id", async (req, res) => {
    const productId = req.params.id;

    // Fetch product details
    let sql = "SELECT * FROM products WHERE product_id = ?";
    let products = await exe(sql, [productId]);

    if (!products || products.length === 0) {
        return res.send("Product not found");
    }

    // Fetch categories for dropdown
    sql = "SELECT * FROM category";
    let categories = await exe(sql);

    res.render("admin/edit_product", { product: products[0], categories });
});

// GET: Add Product Variants page
router.get("/add_product_variants/:product_id", async (req, res) => {
    const { product_id } = req.params;

    const product_result = await exe("SELECT * FROM products WHERE product_id = ?", [product_id]);
    const product = product_result[0];

    const variants = await exe("SELECT * FROM product_variants WHERE product_id = ?", [product_id]);

    res.render("admin/add_product_variants", { product, variants });
});


// POST: Save Product Variants
router.post("/save_product_variants", async (req, res) => {
    const { product_id } = req.body;

    // Arrays
    const titles = Array.isArray(req.body.variation_title) ? req.body.variation_title : [req.body.variation_title];
    const stocks = Array.isArray(req.body.available_stock) ? req.body.available_stock : [req.body.available_stock];
    const market_prices = Array.isArray(req.body.variation_market_price) ? req.body.variation_market_price : [req.body.variation_market_price];
    const prices = Array.isArray(req.body.variation_price) ? req.body.variation_price : [req.body.variation_price];

    // Files
    let files = [];
    if (req.files && req.files.image) {
        files = Array.isArray(req.files.image) ? req.files.image : [req.files.image];
    }

    for (let i = 0; i < titles.length; i++) {
        if (!titles[i] || titles[i].trim() === "") continue; // skip empty

        let filename = "";
        if (files[i]) {
            filename = Date.now() + "_" + files[i].name.replace(/\s/g, "_");
            await files[i].mv("./public/uploads/" + filename);
        }

        await exe(`INSERT INTO product_variants
            (product_id, variation_title, available_stock, variation_market_price, variation_price, image)
            VALUES (?, ?, ?, ?, ?, ?)`, [
            product_id,
            titles[i].trim(),
            stocks[i] || 0,
            market_prices[i] || 0,
            prices[i] || 0,
            filename
        ]);
    }

    res.redirect("/admin/add_product_variants/" + product_id);
});

        slider_button_link

});

// Multiple product images save करण्यासाठी
router.post("/add_product_images/:product_id", async (req, res) => {
  const { product_id } = req.params;

  if (req.files && req.files.product_images) {
    let files = req.files.product_images;

    // जर user ने एक image upload केलं असेल तर files object असेल, array नाही
    if (!Array.isArray(files)) {
      files = [files];
    }

    for (let file of files) {
      let file_name = Date.now() + "_" + file.name.replace(/\s/g, "_");
      await file.mv("./public/uploads/" + file_name);

      const sql = `INSERT INTO product_images (product_id, image_path) VALUES (?, ?)`;
      await exe(sql, [product_id, file_name]);
    }
  }

  res.redirect("/admin/products_list");
});
